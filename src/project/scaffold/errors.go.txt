package @@PACKAGE@@

import "fmt"

type APIError struct { StatusCode int; RawBody []byte }
func (e *APIError) Error() string { return fmt.Sprintf("API error %d: %s", e.StatusCode, e.RawBody) }
func newAPIError(status int, body []byte) *APIError { return &APIError{StatusCode: status, RawBody: body} }
type TransportError struct { Method, Path string; Err error }
func (e *TransportError) Error() string { return fmt.Sprintf("%s %s: %v", e.Method, e.Path, e.Err) }
func (e *TransportError) Unwrap() error { return e.Err }
type DecodeError struct { StatusCode int; RawBody []byte; Err error }
func (e *DecodeError) Error() string { return fmt.Sprintf("decode response %d: %v", e.StatusCode, e.Err) }
func (e *DecodeError) Unwrap() error { return e.Err }
